# ============= TIMESHEET (MATRIX) ============= def _sync_timesheet_matrix(user_id, month, year): with get_conn() as conn: with conn.cursor(cursor_factory=RealDictCursor) as cur: # Определяем права текущего пользователя cur.execute("SELECT * FROM users WHERE user_id = %s", (user_id,)) u = cur.fetchone() if not u: return [] pos_lower = (u.get("position") or "").lower() is_hr = pos_lower.strip() == "hr" is_tu = "территориальный" in pos_lower is_us = "управляющий сменой" in pos_lower # Обычный сотрудник видит только себя query += " AND u.user_id = %s" params.append(user_id) query += " ORDER BY u.full_name" cur.execute(query, params) employees = [dict(r) for r in cur.fetchall()] if not employees: return [] emp_ids = [e["user_id"] for e in employees] # Получаем часы за месяц cur.execute(""" SELECT user_id, work_date, hours FROM timesheet_records WHERE user_id = ANY(%s) AND EXTRACT(MONTH FROM work_date) = %s AND EXTRACT(YEAR FROM work_date) = %s """, (emp_ids, month, year)) records = cur.fetchall() #
